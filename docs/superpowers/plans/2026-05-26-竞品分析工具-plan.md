# 竞品分析工具 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为淘宝店铺诊断工具新增浏览器扩展（数据抓取）+ 竞品分析 Tab + 明日清单 Tab，合并单品诊断能力。

**Architecture:** Chrome/Edge 扩展注入 Content Script 到淘宝搜索页，从渲染后的 DOM 提取商品数据并写入剪贴板。index.html 新增两个 Tab，通过剪贴板读取 JSON 数据，展示 4 张分析卡片和可勾选任务清单。扩展和网页之间通过剪贴板传递结构化数据，扩展也支持在淘宝页上显示浮动按钮供手动触发。

**Tech Stack:** 纯 HTML/CSS/JS（无构建依赖）、Chrome Extension Manifest V3、剪贴板 API、localStorage

### Task 2: 编写 Content Script 数据提取逻辑

**Files:**

- [ ] **Step 1: 编写商品提取函数**

```javascript
// content.js — 注入淘宝搜索页，提取商品数据

function extractProducts() {
  const products = [];

  // 淘宝搜索页商品卡片：多种可能的选择器
  const selectors = [
    '.m-itemlist .item',
    '[data-spm="item"]',
    '.J_MatchItem',
    '.item.J_MatchItem',
    '.grid-view .item'
  ];

  let items = [];
  for (const sel of selectors) {
    items = document.querySelectorAll(sel);
    if (items.length > 0) break;
  }

  // 如果上面都没命中，用更通用的方式：找包含价格和标题的卡片
  if (items.length === 0) {
    const allItems = document.querySelectorAll('[data-nid], [data-item-id]');
    if (allItems.length > 0) {
      items = allItems;
    }
  }

  items.forEach((item, index) => {
    try {
      const product = extractSingleProduct(item, index);
      if (product && product.title && product.price) {
        products.push(product);
      }
    } catch (e) {
      // 跳过解析失败的商品
    }
  });

  return products;
}

function extractSingleProduct(item, index) {
  // 标题：通常在 .title 或 a 标签内
  const titleEl = item.querySelector('.title a, .J_ClickStat, a[title], .row-2 a');
  const title = (titleEl ? titleEl.textContent.trim() || titleEl.getAttribute('title') : '');

  // 价格：通常在 .price 或 strong 标签内
  const priceEl = item.querySelector('.price strong, .price, .c-price, [class*="price"] strong');
  let price = '';
  if (priceEl) {
    price = priceEl.textContent.trim().replace(/[^0-9.]/g, '');
  }

  // 月销量：通常包含"人付款"或"笔"等文字
  const salesEl = item.querySelector('.deal-cnt, .sale-cnt, [class*="deal"], [class*="sale"]');
  let sales = '';
  if (salesEl) {
    sales = salesEl.textContent.trim();
  }

  // 店铺名
  const shopEl = item.querySelector('.shopname, .shop span, [class*="shop"] span, [class*="seller"]');
  let shop = '';
  if (shopEl) {
    shop = shopEl.textContent.trim();
  }

  // 天猫/C店标识
  const isTmall = !!item.querySelector('.tmall, [class*="tmall"], .icon-tmall, img[src*="tmall"]');

  // 活动标签（角标、满减标识等）
  const activityTags = [];
  const tagEls = item.querySelectorAll('.icon, .tag, [class*="icon"], [class*="tag"], .activity');
  tagEls.forEach(el => {
    const text = el.textContent.trim();
    if (text && text.length < 20 && !/^\d+$/.test(text)) {
      activityTags.push(text);
    }
  });

  // 优惠信息（价格下方文案，如"满200减20"）
  const promoEl = item.querySelector('.coupon, [class*="coupon"], [class*="promo"], .yh');
  let promo = '';
  if (promoEl) {
    promo = promoEl.textContent.trim();
  }

  return {
    index: index + 1,
    title,
    price: parseFloat(price) || 0,
    sales,
    shop,
    isTmall,
    activityTags,
    promo
  };
}

// 自动提取并显示浮动按钮
function init() {
  const products = extractProducts();

  if (products.length === 0) return;

  // 创建浮动提示条
  const bar = document.createElement('div');
  bar.id = '__tb_extract_bar__';
  bar.style.cssText = 'position:fixed;top:0;left:0;right:0;z-index:99999;' +
    'background:#ff5000;color:#fff;padding:10px 20px;display:flex;' +
    'align-items:center;justify-content:center;gap:12px;font-size:14px;' +
    'font-family:-apple-system,BlinkMacSystemFont,sans-serif;';

  const text = document.createElement('span');
  text.textContent = `已提取 ${products.length} 个商品数据`;

  const btn = document.createElement('button');
  btn.textContent = '复制到剪贴板';
  btn.style.cssText = 'background:#fff;color:#ff5000;border:none;padding:6px 16px;' +
    'border-radius:4px;cursor:pointer;font-size:13px;font-weight:500;';
  btn.onclick = async () => {
    const json = JSON.stringify({ source: 'taobao-search', url: location.href, count: products.length, products });
    try {
      await navigator.clipboard.writeText(json);
      text.textContent = `已复制 ${products.length} 个商品到剪贴板，请返回工具页面`;
      btn.textContent = '已复制 ✓';
      btn.style.background = '#e8f5e9';
      btn.style.color = '#2e7d32';
    } catch (e) {
      text.textContent = '复制失败，请点击重试';
    }
  };

  bar.appendChild(text);
  bar.appendChild(btn);
  document.body.insertBefore(bar, document.body.firstChild);
}

// 页面加载完成后延迟提取（等动态内容渲染）
setTimeout(init, 2000);
```

```javascript
  const targetTab = document.querySelector('.tab[onclick*="' + tab + '"]');
```

- [ ] **Step 1: 不需要额外的剪贴板自动检测 — 直接实现 paste 区域的监听**

当前 paste 输入方案已经是最可靠的：用户粘贴到 textarea 后点「分析数据」。剪贴板自动读取 API 在 file:// 下受限严重，改为用户的 Ctrl+V + 点击是最小摩擦方案。

实现 paste textarea 的 input 事件：用户粘贴后 textarea 有内容，立即高亮「分析数据」按钮。

在 `</script>` 前添加：

```javascript
// ===== 通用工具 =====
function $(id) { return document.getElementById(id); }

// ===== 竞品分析 =====

// 存储分析结果，供明日清单使用
let lastAnalysisResult = null;
let checklistTasks = JSON.parse(localStorage.getItem('tb_checklist_tasks') || '[]');

function startCompeteSearch() {
  const keyword = $('compKeyword').value.trim();
  if (!keyword) return alert('请输入搜索关键词');

function readFromPaste() {

  if (!raw) return alert('请先在淘宝搜索页点击「复制到剪贴板」，然后回到这里粘贴 JSON 数据');
```

```javascript
function analyzeAndRender(products, myPrice, myCost) {
  // 过滤有效数据（有标题且价格 > 0）
  const valid = products.filter(p => p.title && p.price > 0);

  // --- 标题关键词分析 ---
  const allTitles = valid.map(p => p.title);
  const keywordFreq = getKeywordFrequency(allTitles);
  const topKeywords = keywordFreq.slice(0, 15);

  // --- 价格分析 ---
  const prices = valid.map(p => p.price).sort((a, b) => a - b);
  const avgPrice = prices.length > 0 ? prices.reduce((a, b) => a + b, 0) / prices.length : 0;
  const minPrice = prices[0] || 0;
  const maxPrice = prices[prices.length - 1] || 0;

  // --- 活动分析 ---
  const activityCount = {};
  valid.forEach(p => {
    if (p.activityTags) {
      p.activityTags.forEach(tag => {
        activityCount[tag] = (activityCount[tag] || 0) + 1;
      });
    }
    if (p.promo) {
      const key = '优惠:' + p.promo;
      activityCount[key] = (activityCount[key] || 0) + 1;
    }
  });

  // 按使用次数排序
  const sortedActivities = Object.entries(activityCount)
    .sort((a, b) => b[1] - a[1])
    .slice(0, 10);

  // --- 天猫占比 ---
  const tmallCount = valid.filter(p => p.isTmall).length;

  // 渲染四张卡片
  renderTitleCard(topKeywords, valid.length);
  renderPriceCard(myPrice, avgPrice, minPrice, maxPrice, prices);
  renderActivityCard(sortedActivities, valid.length);
  if (myCost > 0 && myPrice > 0) {
    renderProfitCard(myPrice, myCost, avgPrice, sortedActivities);
  }

  // 渲染数据表
  renderDataTable(valid);

  // 保存分析结果供明日清单使用
  lastAnalysisResult = {
    topKeywords, myPrice, avgPrice, minPrice, maxPrice,
    sortedActivities, valid, tmallCount, totalCount: products.length,
    myCost
  };

  // 生成明日清单任务
  generateChecklistTasks(lastAnalysisResult);
}
```

```javascript
function getKeywordFrequency(titles) {
  // 分词：按空格、逗号、顿号以及常见连接词拆分
  const freq = {};
  const stopWords = ['的', '和', '与', '及', '之', 'a', 'the', 'is', 'for', 'and'];

  titles.forEach(title => {
    // 按常见分隔符拆分
    const words = title.split(/[\s,，、/｜|]+/).filter(w => w.length >= 2);
    // 也尝试提取连续的 2-4 字词组
    const segments = [];
    for (let i = 0; i < title.length - 1; i++) {
      for (let len = 2; len <= 4; len++) {
        const seg = title.substring(i, i + len);
        if (!/[^一-龥a-zA-Z0-9]/.test(seg)) {
          segments.push(seg);
        }
      }
    }
    // 合并拆分结果
    const allWords = [...new Set([...words, ...segments])];
    allWords.forEach(w => {
      if (stopWords.includes(w) || w.length < 2) return;
      freq[w] = (freq[w] || 0) + 1;
    });
  });

  // 排序
  return Object.entries(freq)
    .sort((a, b) => b[1] - a[1])
    .map(([word, count]) => ({ word, count, pct: Math.round(count / titles.length * 100) }));
}
```

```javascript
function generateChecklistTasks(result) {
  if (!result || !result.valid) return;

  const tasks = [];
  const seen = new Set();

  // 规则1: 高频词缺失（用前 10 个高频词作为参考，提示用户检查）
  const topWords = result.topKeywords.slice(0, 10).map(k => k.word).join('、');
  const taskKey1 = 'check_title_keywords';
  if (!seen.has(taskKey1) && result.topKeywords.length >= 3) {
    seen.add(taskKey1);
    tasks.push({
      id: taskKey1,
      text: `检查标题是否包含同行高频词：${topWords}`,
      source: `同行 ${result.topKeywords.length} 个高频词`,
      done: false
    });
  }

  // 规则2: 价格偏离
  if (result.myPrice > 0 && result.avgPrice > 0) {
    const diff = Math.abs((result.myPrice - result.avgPrice) / result.avgPrice * 100);
    if (diff > 30) {
      const taskKey2 = 'adjust_price';
      if (!seen.has(taskKey2)) {
        seen.add(taskKey2);
        tasks.push({
          id: taskKey2,
          text: `考虑调整定价（当前 ¥${result.myPrice}，同行均价 ¥${result.avgPrice.toFixed(0)}，偏离 ${diff.toFixed(0)}%）`,
          source: `价格偏离同行 ${diff.toFixed(0)}%`,
          done: false
        });
      }
    }
  }

  // 规则3: 活动跟进
  if (result.sortedActivities && result.sortedActivities.length > 0) {
    result.sortedActivities.slice(0, 5).forEach(([name, count]) => {
      const pct = Math.round(count / result.valid.length * 100);
      if (pct >= 30) {
        const cleanName = name.replace('优惠:', '');
        const taskId = 'activity_' + cleanName;
        if (!seen.has(taskId)) {
          seen.add(taskId);
          tasks.push({
            id: taskId,
            text: `设置「${cleanName}」活动（去千牛搜索"${cleanName}"查看报名方式）`,
            source: `同行 ${pct}% 使用`,
            done: false
          });
        }
      }
    });
  }

  // 规则4: 运费险
  const hasShipping = result.sortedActivities.some(([name]) => /运费|运费险/.test(name));
  if (hasShipping) {
    const taskKey = 'open_shipping_insurance';
    if (!seen.has(taskKey)) {
      seen.add(taskKey);
      const count = result.sortedActivities.find(([name]) => /运费|运费险/.test(name))[1];
      const pct = Math.round(count / result.valid.length * 100);
      tasks.push({
        id: taskKey,
        text: '开通运费险',
        source: `同行 ${pct}% 支持`,
        done: false
      });
    }
  }

  // 合并已有任务的 done 状态
  const existingMap = {};
  checklistTasks.forEach(t => { existingMap[t.id] = t.done; });
  tasks.forEach(t => {
    if (existingMap.hasOwnProperty(t.id)) {
      t.done = existingMap[t.id];
    }
  });

  checklistTasks = tasks;
  saveChecklist();
  renderChecklist();
}

function saveChecklist() {
  localStorage.setItem('tb_checklist_tasks', JSON.stringify(checklistTasks));

function renderChecklist() {

      `<span>${escHtml(task.text)}</span>` +

      `</div>`;
  });
  container.innerHTML = html;

  // 进度
  const doneCount = checklistTasks.filter(t => t.done).length;
  const total = checklistTasks.length;

  $('checklistProgressText').textContent = `已完成 ${doneCount} / ${total} 项`;
  $('checklistProgressBar').style.width = pct + '%';
```
